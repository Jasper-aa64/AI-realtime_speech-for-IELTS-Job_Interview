cmake_minimum_required(VERSION 3.21)
project(NativeVoiceFoundation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NATIVE_VOICE_QT "Build Qt >=6.11 native callback adapter" ON)
option(NATIVE_VOICE_PORTAUDIO "Build PortAudio callback adapter when available" ON)
set(NATIVE_VOICE_SANITIZER "" CACHE STRING "address, undefined, address,undefined or thread")
if(NATIVE_VOICE_SANITIZER)
    add_compile_options(-fsanitize=${NATIVE_VOICE_SANITIZER} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${NATIVE_VOICE_SANITIZER})
endif()
find_package(Threads REQUIRED)
add_library(native_voice_core src/engine.cpp)
target_include_directories(native_voice_core PUBLIC include)
target_link_libraries(native_voice_core PUBLIC Threads::Threads)
target_compile_options(native_voice_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(native_voice_tests tests/engine_test.cpp)
target_link_libraries(native_voice_tests PRIVATE native_voice_core)
add_executable(native_voice_benchmark src/benchmark.cpp)
target_link_libraries(native_voice_benchmark PRIVATE native_voice_core)
enable_testing()
add_test(NAME native_voice_contracts COMMAND native_voice_tests)
set_tests_properties(native_voice_contracts PROPERTIES TIMEOUT 30)
add_test(NAME native_voice_software_benchmark COMMAND native_voice_benchmark --seconds 0.2)
set_tests_properties(native_voice_software_benchmark PROPERTIES TIMEOUT 10)
if(NATIVE_VOICE_QT)
    find_package(Qt6 6.11 QUIET COMPONENTS Core Multimedia)
    if(Qt6_FOUND)
        add_library(native_voice_qt src/qt_device.cpp)
        target_link_libraries(native_voice_qt PUBLIC native_voice_core Qt6::Core Qt6::Multimedia)
        add_executable(native_voice_qt_hardware src/qt_hardware.cpp)
        target_link_libraries(native_voice_qt_hardware PRIVATE native_voice_qt)
        add_test(NAME native_voice_qt_requires_opt_in COMMAND native_voice_qt_hardware)
        set_tests_properties(native_voice_qt_requires_opt_in PROPERTIES WILL_FAIL TRUE)
    else()
        message(STATUS "Qt >=6.11 Multimedia unavailable; core stays independent")
    endif()
endif()
if(NATIVE_VOICE_PORTAUDIO)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(PORTAUDIO QUIET IMPORTED_TARGET portaudio-2.0)
    endif()
    if(PORTAUDIO_FOUND)
        add_library(native_voice_portaudio src/portaudio_device.cpp)
        target_link_libraries(native_voice_portaudio PUBLIC native_voice_core PkgConfig::PORTAUDIO)
        add_executable(native_voice_portaudio_hardware src/portaudio_hardware.cpp)
        target_link_libraries(native_voice_portaudio_hardware PRIVATE native_voice_portaudio)
        add_test(NAME native_voice_portaudio_requires_opt_in COMMAND native_voice_portaudio_hardware)
        set_tests_properties(native_voice_portaudio_requires_opt_in PROPERTIES WILL_FAIL TRUE)
    else()
        message(STATUS "PortAudio unavailable; optional adapter omitted")
    endif()
endif()
